----------------------------------------**************************--------------------------------------	

There are 3 type of condition that we can do :

Fulfill 
continue 
break

Fulfill : Used to mock the route response and sent it to browser. 

Contine : used to mock the get call and sent another response back to browser. 

break :  Used to block the given request 

Syntax: 
  await page.route(
    "https://rahulshettyacademy.com/api/ecom/order/get-orders-for-customer/*",
    async (route) => {
      const originalResponse = await page.request.fetch(route.request()); // Fetch the original response from the server for the call which in route method and store it in a variable

      route.fulfill({
        // fulfill the route with the original response and modify the response with fakeOrderResponse

        originalResponse,
        body: JSON.stringify(fakeOrderResponse), //Fake response is passed to body to modify the response and validate the order is not displayed on order page
      });
    },
  );


continue: 

  await page.route(
    "https://rahulshettyacademy.com/api/ecom/order/get-orders-details?id=*",
    async (route) => {
      await route.continue({
        url: "https://rahulshettyacademy.com/api/ecom/order/get-orders-details?id=610f3e7c1b0d9a0015e4f8c6",
      });
    },
  );


abort: 

  //Route .abort() is used to abort the request and validate the order is not displayed on order page
  await page.waitForResponse(
    "https://rahulshettyacademy.com/api/ecom/order/get-orders-for-customer/*",
  );
  await page
    .locator('text=" You have No Orders to show at this time."')
    .isVisible();

  await page.route(
    "https://rahulshettyacademy.com/api/ecom/product/get-all-products",
    async (route) => {
      const DashboardOriginalResponse = await page.request.fetch(
        route.request(),
      );
      route.abort(); // Aborting the request to validate the order is not displayed on order page
    },
  );

  await page.getByRole("button", { name: "Home" }).click({ delay: 500 });
  await page.locator(".container").waitFor();
  const validation = await page
    .getByRole("heading", { name: "ADIDAS ORIGINAL" })

const myfixtures=require('@playwright/test');  // Rather than calling test , I have named as myfixtures

exports.customtest=myfixtures.test.extend( // Here i have extended my custom fixtures to use test fixtures capabilites
